export async function handleChatMessage(db, event, generate) {
  const message = event.data?.data();
  if (message?.senderRole !== 'member' || typeof message.text !== 'string') return;
  const { uid, messageId } = event.params;
  const chatRef = db.doc(`memberChats/${uid}`);
  const replyRef = chatRef.collection('messages').doc(`ai-${messageId}`);
  const lease = randomUUID();
  const acquired = await db.runTransaction(async tx => {
    const [chatSnap, replySnap] = await Promise.all([tx.get(chatRef), tx.get(replyRef)]);
    const chat = chatSnap.data();
    if (!chat || replySnap.exists || chat.status !== 'waiting' || chat.aiEscalated === true ||
        !chat.lastMemberMessageAt?.isEqual(message.createdAt)) return null;
    if (chat.aiLeaseUntil?.toMillis() > Date.now()) throw new Error('AI lease busy; retry event.');
    const minute = Math.floor(Date.now() / 60000);
    const count = chat.aiMinute === minute ? chat.aiCount || 0 : 0;
    if (count >= 6) {
      tx.update(chatRef, { aiEscalated: true, aiError: 'rate_limited' });
      return null;
    }
    tx.update(chatRef, { aiLease: lease, aiLeaseUntil: Timestamp.fromMillis(Date.now() + 60000), aiMinute: minute, aiCount: count + 1 });
    return { epoch: chat.aiEpoch || 0 };
  });
  if (!acquired) return;
  let result;
  try {
    const history = await chatRef.collection('messages').orderBy('createdAt', 'desc').limit(12).get();
    result = await generate(history.docs.reverse().filter(d => d.id !== 'support-ack').map(d => {
      const m = d.data();
      return { role: m.senderRole === 'member' ? 'user' : 'assistant', text: String(m.text || '').slice(0, 4000) };
    }));
    if (typeof result?.text !== 'string' || !result.text.trim()) throw new Error('Empty provider reply');
  } catch {
    result = { text: FALLBACK, escalate: true };
  }
  await db.runTransaction(async tx => {
    const [chatSnap, replySnap] = await Promise.all([tx.get(chatRef), tx.get(replyRef)]);
    const chat = chatSnap.data();
    if (!chat || replySnap.exists || chat.aiLease !== lease) return;
    tx.update(chatRef, { aiLeaseUntil: Timestamp.fromMillis(0) });
    // Epoch rejects stale generations even after a takeover/release cycle.
    if (chat.status !== 'waiting' || (chat.aiEpoch || 0) !== acquired.epoch ||
        chat.aiEscalated === true || !chat.lastMemberMessageAt?.isEqual(message.createdAt)) return;
    tx.create(replyRef, { senderId: 'asbesoc-ai', senderRole: 'system',
      text: `${chat.aiIntroduced ? '' : INTRO}${result.text.trim()}`.slice(0, 4000), createdAt: FieldValue.serverTimestamp() });
    tx.update(chatRef, { aiIntroduced: true, aiEscalated: result.escalate === true,
      aiError: result.escalate ? 'administrator_required' : null, updatedAt: FieldValue.serverTimestamp() });
  });
}

      ) {
        if (previousOrder.currency !== currency) {
          throw new HttpsError(
            'failed-precondition',
            'An existing checkout uses another currency. Please wait for it to expire.'
          );
        }

        if (previousOrder.amountMinor !== amountMinor) {
          throw new HttpsError(
            'failed-precondition',
            'The payment amount has changed. Please contact ASBESOC.'
          );
        }

        return previousOrder;
      }

      // No usable pending order: create a fresh one.
      const reference = newReference;

      const value = {
        reference,
        uid,
        email: request.auth.token.email,
        fullName: application.fullName,
        amountMinor,
        currency,
        status: 'pending',
        createdAt: Timestamp.now(),
      };

      tx.create(
        db.doc(`paymentOrders/${reference}`),
        value
      );

      tx.set(lockRef, {
        reference,
        expiresAt: Timestamp.fromMillis(
          now + CHECKOUT_LOCK_MINUTES * 60000
        ),
      });

      return value;
    });

    // Reuse a checkout URL when already initialized.
    if (order.authorizationUrl) {
      return {
        url: validateCheckoutUrl(
          order.authorizationUrl
        ),
        reference: order.reference,
      };
    }

    // Paystack initialization uses the same reference.
    // Retrying initialization must not create another order.
    const result = await paystack(
      '/transaction/initialize',
      {
        email: order.email,
        amount: order.amountMinor,
        currency: order.currency,
        reference: order.reference,
        metadata: {
          uid: order.uid,
        },
        callback_url:
          'https://asbesoc.org/dashboard/membership',
      }
    );

    const checkoutUrl = validateCheckoutUrl(
      result.authorization_url
    );

    await db.runTransaction(async tx => {
      const orderRef = db.doc(
        `paymentOrders/${order.reference}`
      );

      const snapshot = await tx.get(orderRef);

      if (!snapshot.exists) {
        throw new HttpsError(
          'not-found',
          'Payment order not found.'
        );
      }

      if (snapshot.data().status !== 'pending') {
        return;
      }

      if (!snapshot.data().authorizationUrl) {
        tx.update(orderRef, {
          authorizationUrl: checkoutUrl,
        });
      }
    });

    return {
      url: checkoutUrl,
      reference: order.reference,
    };
  }
);

async function settle(reference, owner) {
  await paymentConfig();

  const orderRef = db.doc(
    `paymentOrders/${validId(reference)}`
  );

  const snapshot = await orderRef.get();

  if (
    !snapshot.exists ||
    (owner && snapshot.data().uid !== owner)
  ) {
    throw new HttpsError(
      'not-found',
      'Payment order not found.'
    );
  }

  if (snapshot.data().status === 'paid') {
    return { confirmed: true };
  }

  const data = await paystack(
    `/transaction/verify/${encodeURIComponent(reference)}`
  );

  let outcome;
  try {
    outcome = paymentOutcome(data, snapshot.data());
  } catch {
    throw new HttpsError(
      'failed-precondition',
      'Payment is not confirmed. No membership was granted.'
    );
  }

  if (outcome !== 'paid') {
    const status = await db.runTransaction(async tx => {
      const current = await tx.get(orderRef);
      if (!current.exists) throw new HttpsError('not-found', 'Payment order not found.');
      const nextStatus = paymentStatusAfterCheck(current.data().status, outcome);
      if (nextStatus !== current.data().status || nextStatus === outcome) {
        tx.update(orderRef, {
          status: nextStatus,
          providerStatus: data.status,
          lastCheckedAt: FieldValue.serverTimestamp(),
        });
      }
      return nextStatus;
    });
    return { confirmed: status === 'paid', status };
  }

  const certificateId = `ASBESOC-${randomUUID()}`;

  await db.runTransaction(async tx => {
    const orderSnapshot = await tx.get(orderRef);

    if (!orderSnapshot.exists) {
      throw new HttpsError(
        'not-found',
        'Payment order not found.'
      );
    }

    const order = orderSnapshot.data();

    if (order.status === 'paid') {
      return;
    }

    const memberRef = db.doc(
      `memberships/${order.uid}`
    );

    const appRef = db.doc(
      `membershipApplications/${order.uid}`
    );

    const receiptRef = db.doc(
      `providerReceipts/paystack-${data.id}`
    );

    const [appSnapshot, memberSnapshot, receiptSnapshot] =
      await Promise.all([
        tx.get(appRef),
        tx.get(memberRef),
        tx.get(receiptRef),
      ]);

    const application = appSnapshot.data();
    const member = memberSnapshot.data();

    if (receiptSnapshot.exists) {
      throw new HttpsError(
        'already-exists',
        'Transaction already processed.'
      );
    }

    if (
      application?.status !== 'approved' ||
      application?.userId !== order.uid ||
      member?.status === 'revoked'
    ) {
      throw new HttpsError(
        'failed-precondition',
        'Payment needs administrator review.'
      );
    }

    const now = new Date();

    const period = annualPeriod(
      now,
      member?.expiresAt?.toDate()
    );

    // The new certificate covers the renewed period.
    const certificate = {
      id: certificateId,
      uid: order.uid,
      fullName: application.fullName,
      issuedAt: Timestamp.fromDate(now),
      startsAt: Timestamp.fromDate(
        period.startsAt
      ),
      expiresAt: Timestamp.fromDate(
        period.expiresAt
      ),
      status: 'active',
      paymentReference: reference,
    };

    tx.create(
      db.doc(`certificates/${certificateId}`),
      certificate
    );

    // Preserve the original membership start date.
    // Renewal extends the expiry date.
    tx.set(memberRef, {
      uid: order.uid,
      fullName: application.fullName,
      status: 'active',
      certificateId,
      startsAt:
        member?.startsAt ||
        Timestamp.fromDate(now),
      expiresAt: certificate.expiresAt,
      updatedAt: Timestamp.fromDate(now),
    });

    tx.update(appRef, {
      paymentStatus: 'paid',
      paidAt: Timestamp.fromDate(now),
      paymentReference: reference,
      certificateStatus: 'issued',
      certificateNumber: certificateId,
      certificateIssuedAt: certificate.issuedAt,
      membershipExpiresAt: certificate.expiresAt,
    });

    tx.update(orderRef, {
      status: 'paid',
      providerId: data.id,
      confirmedAt: Timestamp.fromDate(now),
      certificateId,
    });

    tx.create(receiptRef, {
      reference,
      uid: order.uid,
      createdAt: Timestamp.fromDate(now),
    });

    tx.create(
      db.doc(`membershipAudit/${certificateId}`),
      {
        type:
          'payment_verified_certificate_issued',
        uid: order.uid,
        reference,
        certificateId,
        createdAt: Timestamp.fromDate(now),
        actor: 'paystack-verification',
      }
    );
  });

  return { confirmed: true };
}

export const paystackWebhook = onRequest(
  {
    secrets: [secret],
  },
  async (req, res) => {
    if (req.method !== 'POST') {
      res.status(405).send('Method not allowed');
      return;
    }

    if (
      !verifySignature(
        req.rawBody,
        req.get('x-paystack-signature'),
        secret.value()
      )
    ) {
      res.status(401).send('Invalid signature');
      return;
    }

    if (req.body?.event !== 'charge.success') {
      res.status(200).send('Ignored');
      return;
    }

    try {
      const result = await settle(req.body.data?.reference);
      if (!result.confirmed) { res.status(503).send('Not confirmed; retry required'); return; }
      res.status(200).send('Received');
    } catch {
      res.status(503).send(
        'Not processed; retry required'
      );
    }
  }
);

export const verifyMembershipCertificate = onCall(
  {
    enforceAppCheck: true,
  },
  async request => {
    const id = validId(request.data?.id);

    const certificate = (
      await db.doc(`certificates/${id}`).get()
    ).data();

    if (!certificate) {
      return { status: 'not_found' };
    }

    const member = (
      await db.doc(
        `memberships/${certificate.uid}`
      ).get()
    ).data();

    if (
      !member ||
      !['active', 'revoked'].includes(
        member.status
      ) ||
      !['active', 'revoked'].includes(
        certificate.status
      )
    ) {
      return { status: 'not_found' };
    }

    const now = Date.now();

    const status =
      certificate.status === 'revoked' ||
      member.status === 'revoked'
        ? 'revoked'
        : certificate.expiresAt.toMillis() <= now ||
            member.expiresAt?.toMillis() <= now
          ? 'expired'
          : certificate.startsAt.toMillis() > now ||
              member.startsAt?.toMillis() > now
            ? 'not_yet_valid'
            : activeMembership(member)
              ? 'valid'
              : 'not_found';

    return {
      id,
      fullName: certificate.fullName,
      status,
      issuedAt:
        certificate.issuedAt
          .toDate()
          .toISOString(),
      startsAt:
        certificate.startsAt
          .toDate()
          .toISOString(),
      expiresAt:
        certificate.expiresAt
          .toDate()
          .toISOString(),
    };
  }
);

      );

      const member = await tx.get(memberRef);

        throw new HttpsError(

        return;

        }

        db.collection('membershipAudit').doc(),
        {
          type: 'revoked',
          uid,
          reason,
          actor,
          createdAt:
            FieldValue.serverTimestamp(),
        }
      );
    });

    return { revoked: true };
  }
